```java
public class BatchExtendCmd {
    @NotNull Long planId;
```

- [ ] **Step 2: 实现 service.batchExtend**

```java
@Transactional(rollbackFor = Exception.class)
public void batchExtend(BatchExtendCmd cmd) {
    // 1. 校验计划存在
    TrainingPlan plan = trainingPlanMapper.selectById(cmd.getPlanId());
    if (plan == null) throw new BizException("计划不存在");
    // 2. 校验日期 >= 计划当天(硬约束,docx 原话)
    Date planStart = plan.getStartDate();
    if (cmd.getNewEndDate().before(planStart)) {
        throw new BizException("延期时间必须 >= 计划当天");
    }
    // 3. 批量更新学员的 el_training_record.plan_end_date
    for (Long userId : cmd.getUserIds()) {
        trainingRecordMapper.extendPlanEndDate(cmd.getPlanId(), userId, cmd.getNewEndDate());
        // 4. 写流水(若有 el_training_record_log 表)
        extendLogMapper.insert(ExtendLog.builder()
            .planId(cmd.getPlanId()).userId(userId)
            .newEndDate(cmd.getNewEndDate()).reason(cmd.getReason())
            .operator(SecurityUtils.getCurrentUserId())
            .createTime(new Date()).build());
    }
}
```

- [ ] **Step 3: 写 mapper.xml 的 SQL**

```xml
<update id="extendPlanEndDate">

    SET plan_end_date = #{newEndDate}, update_time = NOW()

</update>
```

```java
@Transactional(rollbackFor = Exception.class)
public void batchMakeup(BatchMakeupCmd cmd) {
    // 1. 校验计划存在
    TrainingPlan plan = trainingPlanMapper.selectById(cmd.getPlanId());
    if (plan == null) throw new BizException("计划不存在");
    // 2. 硬约束:补考时间不能为空(docx:补考必带延期)
    if (cmd.getMakeupDate() == null) throw new BizException("补考必须设置时间");
    // 3. 校验 makeupDate >= 计划当天(同延期硬约束)
    if (cmd.getMakeupDate().before(plan.getStartDate())) {
        throw new BizException("补考时间必须 >= 计划当天");
    }
    // 4. 批量处理每条未通过考试
    for (Long examRecordId : cmd.getExamRecordIds()) {
        ExamRecord er = examRecordMapper.selectById(examRecordId);
        if (er == null || er.getStatus() == ExamStatus.PASSED) continue; // 已通过的跳过
        // 4.1 记录补考次数 + 补考时间
        examRecordMapper.recordMakeup(examRecordId, cmd.getMakeupDate());
        // 4.2 **同时**延长学员计划结束时间(硬约束:补考必带延期)
        trainingRecordMapper.extendPlanEndDate(cmd.getPlanId(), er.getUserId(), cmd.getMakeupDate());
        // 4.3 写补考流水
        makeupLogMapper.insert(MakeupLog.builder()
            .examRecordId(examRecordId).userId(er.getUserId())
            .makeupDate(cmd.getMakeupDate()).reason(cmd.getReason())
            .operator(SecurityUtils.getCurrentUserId()).createTime(new Date()).build());
    }
}
```
