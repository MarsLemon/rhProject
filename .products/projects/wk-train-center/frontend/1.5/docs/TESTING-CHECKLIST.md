```sql
-- =====================================================
-- 测试环境初始化脚本（根据实际情况调整）
-- =====================================================

-- 1. 创建测试学习计划
INSERT INTO el_plan (id, title, start_time, end_time, state, create_time)
VALUES ('PLN_TEST_001', '延期功能测试计划', NOW(), DATE_ADD(NOW(), INTERVAL 7 DAY), 'NORMAL', NOW())
ON DUPLICATE KEY UPDATE id=id;

-- 2. 创建测试学员账号
INSERT INTO yf_user (user_name, password, real_name, state, email)
VALUES 
  ('admin_test', 'e10adc3949ba59abbe56e057f20f883e', '测试管理员', 'NORMAL', 'admin@test.com'),
  ('studentA_test', 'e10adc3949ba59abbe56e057f20f883e', '学员 A 测试', 'NORMAL', 'studentA@test.com')
ON DUPLICATE KEY UPDATE user_name=user_name;

-- 3. 为学员创建未完成的学习记录
INSERT INTO el_plan_user (plan_id, user_id, state, deadline, extend_deadline_count)
SELECT 
  'PLN_TEST_001' AS plan_id,
  CONCAT('U', u.id) AS user_id,  -- 根据实际 userId 格式调整
  'INCOMPLETE' AS state,
  DATE_SUB(NOW(), INTERVAL 1 DAY) AS deadline,
  0 AS extend_deadline_count
FROM yf_user u
WHERE u.user_name IN ('admin_test', 'studentA_test')
ON DUPLICATE KEY UPDATE plan_id=plan_id;

-- 4. 插入模拟考试未通过记录
INSERT INTO el_exam_record (user_id, exam_id, score, passed, create_time)
SELECT 
  CONCAT('U', u.id),
  'EXAM_001',  -- 假设已有考试 ID
  65 AS score,
  0 AS passed,
  NOW() AS create_time
FROM yf_user u
WHERE u.user_name = 'studentA_test'
ON DUPLICATE KEY UPDATE user_id=user_id;

-- 5. 验证查询（应返回准备好的数据）
SELECT 
  p.title AS plan_title,
  u.user_name AS student_name,
  pup.state,
  pup.deadline,
  pup.extend_deadline_count
FROM el_plan p
JOIN el_plan_user pup ON p.id = pup.plan_id
JOIN yf_user u ON SUBSTRING_INDEX(u.id, '_', 1) = SUBSTRING_INDEX(pup.user_id, 'U', 1)
WHERE p.id = 'PLN_TEST_001';
```
